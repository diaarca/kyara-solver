cmake_minimum_required(VERSION 3.24)
project(kyara_solver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ortools CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(solve src/main.cpp)
target_link_libraries(solve PRIVATE ortools::ortools nlohmann_json::nlohmann_json)
target_compile_definitions(solve PRIVATE KYARA_ORTOOLS_VERSION="${ORTOOLS_VERSION}")
target_compile_options(solve PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS solve RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME workflow COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
  set_tests_properties(workflow PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    ENVIRONMENT "PYTHONPATH=${CMAKE_SOURCE_DIR}/python;KYARA_SOLVER=$<TARGET_FILE:solve>")
endif()
